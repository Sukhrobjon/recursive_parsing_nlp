```python
import pandas as pd

# Load the two CSV source files
df_summer = pd.read_csv('./Athletes_summer_games.csv')
df_winter = pd.read_csv('./Athletes_winter_games.csv')

# Create an empty DataFrame that will hold the counts
total_games = pd.DataFrame(columns=['Season', 'Total # of Games'])

# Calculate unique Games counts for each season
winter_count = df_winter['Games'].nunique()
summer_count = df_summer['Games'].nunique()

# Populate the summary table
total_games.loc[0] = ['Winter', winter_count]
total_games.loc[1] = ['Summer', summer_count]

# Save the result to the required output file
total_games_path = './allGames.csv'
total_games.to_csv(total_games_path, index=False)

# Display the final DataFrame so we can verify interactively
total_games
```
