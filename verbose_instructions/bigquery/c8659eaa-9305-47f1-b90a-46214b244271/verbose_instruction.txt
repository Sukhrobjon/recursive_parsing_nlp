8. Highlight any placeholder code in query.py, press Delete, and type (or paste) the complete runnable code below:  
```python
# query.py
import os
import pandas as pd
from pathlib import Path
from google.cloud import bigquery

def query_data() -> None:
    # Point to your service-account key; adjust the filename if needed
    os.environ["GOOGLE_APPLICATION_CREDENTIALS"] = "credential.json"

    client = bigquery.Client()

    # SQL: compute floor area then sort descending
    query = """
        SELECT
          name,
          (footprint_length * footprint_width) AS area
        FROM
          `bigquery-public-data.austin_bikeshare.bikeshare_stations`
        ORDER BY
          area DESC
        LIMIT 10
    """
    # Run query
    query_job = client.query(query)
    results_df = query_job.result().to_dataframe()

    # Ensure the project1 folder exists locally
    Path("project1").mkdir(exist_ok=True)

    # Save with required headers name, area
    results_df.to_csv("project1/result.csv", index=False)
    print("✅ result.csv with columns [name, area] written to project1/")

if __name__ == "__main__":
    query_data()
```
