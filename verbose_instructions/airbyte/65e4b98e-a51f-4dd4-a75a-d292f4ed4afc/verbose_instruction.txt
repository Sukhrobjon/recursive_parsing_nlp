```python
from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.providers.mysql.hooks.mysql import MySqlHook
from datetime import datetime, timedelta
import json, os

# ---- default arguments ----
default_args = {
    "owner": "data_eng",
    "depends_on_past": False,
    "retries": 1,
    "retry_delay": timedelta(minutes=5),
}

# ---- constants we will reuse ----
_CONN_ID = "mysql_sync_conn"
_OUTPUT_DIR = "/json_data"            # make sure this exists on the host
_TABLES = ["orders", "customers", "payments"]

# ---- utility function ----
def export_tables_to_json(**context):
    """Export each table listed in _TABLES to a separate JSON file."""
    hook = MySqlHook(mysql_conn_id=_CONN_ID)
    conn = hook.get_conn()
    cursor = conn.cursor(dictionary=True)
    os.makedirs(_OUTPUT_DIR, exist_ok=True)

    for table in _TABLES:
        cursor.execute(f"SELECT * FROM {table};")
        rows = cursor.fetchall()
        output_path = os.path.join(_OUTPUT_DIR, f"{table}.json")
        with open(output_path, "w", encoding="utf-8") as f:
            json.dump(rows, f, default=str, indent=2)
        print(f"Wrote {len(rows)} rows from {table} to {output_path}")

    cursor.close()
    conn.close()

# ---- DAG definition ----
with DAG(
    dag_id="mysql_to_local_json",
    description="Sync MySQL tables to local JSON files every 12 hours",
    start_date=datetime(2023, 1, 1),
    schedule_interval="0 */12 * * *",   # every 12 hours at minute 0
    catchup=False,
    default_args=default_args,
    max_active_runs=1,
    tags=["mysql", "json", "sync"],
) as dag:

    dump_to_json = PythonOperator(
        task_id="dump_mysql_to_json",
        python_callable=export_tables_to_json,
    )

    dump_to_json
```
