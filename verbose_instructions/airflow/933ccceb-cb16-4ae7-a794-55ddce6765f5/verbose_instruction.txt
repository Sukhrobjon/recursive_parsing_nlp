```python
from airflow import DAG
from airflow.providers.common.sql.operators.sql import SQLExecuteQueryOperator
from airflow.providers.common.sql.operators.sql_check import (
    SQLColumnCheckOperator,
    SQLTableCheckOperator,
)
from airflow.utils.dates import days_ago

_CONN_ID = "sqlite_conn"
_TABLE_NAME = "birds"

with DAG(
    dag_id="sql_data_quality",
    start_date=days_ago(1),
    schedule_interval=None,
    catchup=False,
    tags=["examples", "sqlite", "data_quality"],
) as dag:

    # 1) Build a demo table and populate test data
    init_db = SQLExecuteQueryOperator(
        task_id="init_db",
        conn_id=_CONN_ID,
        sql="""
        DROP TABLE IF EXISTS birds;
        CREATE TABLE birds (
            bird_name TEXT,
            observation_year INTEGER,
            bird_happiness INTEGER
        );
        INSERT INTO birds (bird_name, observation_year, bird_happiness) VALUES
            ('Sparrow', 2019, 9),
            ('Robin',   2020, 8),
            ('Eagle',   2021, 10),
            ('Parrot',  2022, 7),
            ('Crow',    2023, 6),
            ('Finch',   2023, 9);
        """,
    )

    # 2) Column-level checks
    column_checks = SQLColumnCheckOperator(
        task_id="column_checks",
        conn_id=_CONN_ID,
        table=_TABLE_NAME,
        partition_clause="bird_name IS NOT NULL",
        column_mapping={
            "bird_name": {
                "null_check": {"equal_to": 0},
                "distinct_check": {"geq_to": 5},
            },
            "observation_year": {
                "min": {"geq_to": 2018},
                "max": {"less_than": 2024},
            },
            "bird_happiness": {
                "min": {"greater_than": 0},
                "max": {"leq_to": 10},
            },
        },
    )

    # 3) Table-level checks
    table_checks = SQLTableCheckOperator(
        task_id="table_checks",
        conn_id=_CONN_ID,
        table=_TABLE_NAME,
        checks={
            "average_happiness_check": {
                "check_statement": "AVG(bird_happiness) >= 8",
                "partition_clause": "observation_year >= 2019",
            },
        },
    )

    init_db >> column_checks >> table_checks
```
