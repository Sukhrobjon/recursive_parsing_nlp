11) Write the custom check query exactly as below – it will fail if fewer than 20 distinct birds exist after 2019 or if average happiness < 8:

```sql
-- dags/custom_check.sql
/* Custom data-quality assertion
   1) At least 20 distinct birds observed since 2020
   2) Average happiness of those birds must be >= 8
   A single row returning 1 means “pass”, anything else means “fail”.
*/
SELECT
    CASE
        WHEN COUNT(DISTINCT bird_name) >= 20
         AND AVG(bird_happiness) >= 8
        THEN 1
        ELSE 0
    END AS quality_passed
FROM observations
WHERE observation_year >= 2020;
```

```python
# dags/sql_data_quality.py
from datetime import datetime
from airflow import DAG
from airflow.providers.sqlite.operators.sqlite import SqliteOperator
from airflow.providers.common.sql.operators.sql import SQLCheckOperator
from pathlib import Path

_CONN_ID = "sqlite_conn"
_DB_TABLE = "observations"
_SQL_DIR = Path(__file__).parent

default_args = {"retries": 0}

with DAG(
    dag_id="sql_data_quality",
    start_date=datetime(2024, 1, 1),
    schedule=None,
    catchup=False,
    default_args=default_args,
    tags=["quality", "sqlite"],
) as dag:

    # 1. Create table if it does not exist
    create_table = SqliteOperator(
        task_id="create_table",
        sqlite_conn_id=_CONN_ID,
        sql=f"""
        CREATE TABLE IF NOT EXISTS {_DB_TABLE} (
            bird_name TEXT,
            observation_year INTEGER,
            bird_happiness INTEGER
        );
        """,
    )

    # 2. Seed demo data – in a real case you would remove duplicates
    seed_data = SqliteOperator(
        task_id="seed_data",
        sqlite_conn_id=_CONN_ID,
        sql=f"""
        INSERT INTO {_DB_TABLE} (bird_name, observation_year, bird_happiness)
        VALUES
            ('sparrow', 2020, 9),
            ('eagle',   2021, 8),
            ('crow',    2022, 8),
            ('owl',     2023, 9),
            ('finch',   2020, 8),
            ('parrot',  2021, 10),
            ('penguin', 2022, 9),
            ('falcon',  2023, 8),
            ('robin',   2022, 9),
            ('lark',    2023, 9),
            ('swan',    2020, 8),
            ('goose',   2021, 8),
            ('wren',    2022, 9),
            ('hawk',    2023, 8),
            ('dove',    2021, 9),
            ('heron',   2022, 8),
            ('crane',   2023, 9),
            ('pigeon',  2023, 8),
            ('stork',   2021, 9),
            ('kiwi',    2020, 8);
        """,
    )

    # 3. Custom SQL quality check using external file
    custom_quality_check = SQLCheckOperator(
        task_id="custom_quality_check",
        conn_id=_CONN_ID,
        sql=str(_SQL_DIR / "custom_check.sql"),
        check_supports_autocommit=False,
    )

    create_table >> seed_data >> custom_quality_check
```
