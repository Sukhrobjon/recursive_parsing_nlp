    dag = dag_bag.get_dag("activity_suggestion_dag")

7. Type the second required function “test_task_workflow_order” to validate the precise task sequence:  
```python
def test_task_workflow_order():
    """
    Ensure tasks are ordered fetch_activity -> log_activity -> analyze_activity.
    """
    dag = dag_bag.get_dag("activity_suggestion_dag")
    assert dag is not None, "DAG 'activity_suggestion_dag' not found."
    # topological_sort returns tasks in execution order
    task_ids = [task.task_id for task in dag.topological_sort()]
    expected_order = ["fetch_activity", "log_activity", "analyze_activity"]
    assert task_ids == expected_order, (
        f"Tasks are not in the expected order. Expected {expected_order}, got {task_ids}"
    )
```
