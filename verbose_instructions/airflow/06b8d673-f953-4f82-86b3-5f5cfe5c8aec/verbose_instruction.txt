```python
from airflow import DAG
from airflow.providers.sqlite.operators.sqlite import SqliteOperator
from airflow.providers.common.sql.operators.sql import SQLColumnCheckOperator, SQLTableCheckOperator
from airflow.utils.dates import days_ago
from datetime import timedelta

_CONN_ID = "sqlite_conn"
_TABLE_NAME = "bird_observations"

default_args = {
    "owner": "airflow",
    "retries": 1,
    "retry_delay": timedelta(minutes=2),
}

with DAG(
    dag_id="sql_data_quality",
    default_args=default_args,
    start_date=days_ago(1),
    schedule_interval=None,
    catchup=False,
    tags=["example", "data-quality"],
) as dag:

    # 1. Create table if not exists
    create_table = SqliteOperator(
        task_id="create_table",
        sql=f"""
        CREATE TABLE IF NOT EXISTS {_TABLE_NAME} (
            bird_name TEXT,
            observation_year INTEGER,
            bird_happiness INTEGER
        );
        """,
        sqlite_conn_id=_CONN_ID,
    )

    # 2. Populate sample data
    insert_rows = SqliteOperator(
        task_id="insert_rows",
        sql=f"""
        DELETE FROM {_TABLE_NAME};
        INSERT INTO {_TABLE_NAME} (bird_name, observation_year, bird_happiness)
        VALUES
            ('sparrow', 2019, 9),
            ('robin',   2020, 8),
            ('eagle',   2021, 10),
            ('owl',     2021, 7),
            ('crow',    2022, 8);
        """,
        sqlite_conn_id=_CONN_ID,
    )

    # 3. Column-level checks
    column_checks = SQLColumnCheckOperator(
        task_id="column_checks",
        conn_id=_CONN_ID,
        table=_TABLE_NAME,
        column_mapping={
            "bird_name": {
                "null_check": {"equal_to": 0},
                "distinct_check": {"geq_to": 5},
            },
            "observation_year": {
                "min": {"greater_than": 2018},
                "max": {"less_than": 2024},
            },
            "bird_happiness": {
                "min": {"geq_to": 1},
                "max": {"leq_to": 10},
            },
        },
    )

    # 4. Table-level checks
    table_checks = SQLTableCheckOperator(
        task_id="table_checks",
        conn_id=_CONN_ID,
        table=_TABLE_NAME,
        checks={
            "avg_happiness_after_2018": {
                "check_statement": "AVG(bird_happiness) >= 8",
                "partition_clause": "observation_year > 2018",
            },
        },
    )

    create_table >> insert_rows >> column_checks >> table_checks
```
