5) Type the following complete asset definition exactly as shown:

```python
import os
import urllib.request
import json

# Ensure output directory exists
dir_name = "data"
if not os.path.exists(dir_name):
    os.makedirs(dir_name)

from dagster import AssetExecutionContext, DailyPartitionsDefinition, asset

@asset(partitions_def=DailyPartitionsDefinition(start_date="2024-04-01"))
def apod_asset(context: AssetExecutionContext) -> None:
    """
    Partitioned Dagster asset that fetches NASA's Astronomy Picture of the Day
    for each partition date and saves the HD image URL to a text file.
    """
    partition_date_str = context.partition_key  # e.g. '2024-04-10'

    api_url = (
        f"https://api.nasa.gov/planetary/apod"
        f"?api_key=DEMO_KEY&date={partition_date_str}"
    )
    json_target = f"data/{partition_date_str}.json"

    # Download the JSON metadata for the date
    urllib.request.urlretrieve(api_url, json_target)

    # Load JSON to extract the hdurl field
    with open(json_target, "r") as f:
        data = json.load(f)

    # Write the HD image URL to a dated txt file
    txt_target = f"data/apod_url_{partition_date_str}.txt"
    with open(txt_target, "w") as out_f:
        out_f.write(data["hdurl"])
```
