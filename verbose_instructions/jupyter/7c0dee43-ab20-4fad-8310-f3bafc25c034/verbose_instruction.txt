6) Write the following code into the new cell to load the CSVs into SQLite tables (this keeps everything reproducible).  

```python
import pandas as pd, sqlite3, os

# Load raw CSVs
products   = pd.read_csv('products.csv')
categories = pd.read_csv('categories.csv')   if os.path.exists('categories.csv')   else pd.DataFrame()
customers  = pd.read_csv('customers.csv')    if os.path.exists('customers.csv')    else pd.DataFrame()
order_items= pd.read_csv('order_items.csv')  if os.path.exists('order_items.csv')  else pd.DataFrame()
orders     = pd.read_csv('orders.csv')       if os.path.exists('orders.csv')       else pd.DataFrame()
stocks     = pd.read_csv('stocks.csv')       if os.path.exists('stocks.csv')       else pd.DataFrame()
stores     = pd.read_csv('stores.csv')       if os.path.exists('stores.csv')       else pd.DataFrame()

# Create / connect to an in-memory SQLite DB
connection = sqlite3.connect('bike_store.db')

# Persist tables for SQL querying
categories.to_sql('categories', connection, if_exists='replace', index=False)
customers.to_sql('customers', connection, if_exists='replace', index=False)
order_items.to_sql('order_items', connection, if_exists='replace', index=False)
orders.to_sql('orders', connection, if_exists='replace', index=False)
products.to_sql('products', connection, if_exists='replace', index=False)
stocks.to_sql('stocks', connection, if_exists='replace', index=False)
stores.to_sql('stores', connection, if_exists='replace', index=False)
```

9) Type the SQL query and export logic exactly as shown:  

```python
query = """
SELECT *
FROM products
WHERE model_year = 2019
  AND list_price > (
        SELECT AVG(list_price)
        FROM products
        WHERE model_year = 2019
      )
ORDER BY list_price DESC;
"""

df = pd.read_sql_query(query, connection)

# Save to CSV in the same shape as the previewed template
df.to_csv('result.csv', index=False)
print(f"Rows written: {len(df)}")
```
