```python
# keep any existing imports
from airflow.utils.context import Context
from airflow.models.baseoperator import BaseOperatorLink
from airflow.providers.http.operators.http import SimpleHttpOperator


class WTTRWeatherLink(BaseOperatorLink):
    name = "WTTRWeather"           # text that will appear in the UI
    operators = [SimpleHttpOperator]

    # optional ti_key retained for Airflow 2.7+ signature compatibility
    def get_link(self, operator: SimpleHttpOperator, *, ti_key: "TaskInstanceKey" | None = None) -> str:  # type: ignore[name-defined]
        return "https://wttr.in/HongKong"


class HTTPDocsLink(BaseOperatorLink):
    name = "HTTP file"
    operators = [SimpleHttpOperator]

    def get_link(self, operator: SimpleHttpOperator, *, ti_key: "TaskInstanceKey" | None = None) -> str:  # type: ignore[name-defined]
        return "https://developer.mozilla.org/en-US/docs/Web/HTTP"


# If not already present, make sure to register the links so Airflow loads them
LINKS = [WTTRWeatherLink(), HTTPDocsLink()]
```
